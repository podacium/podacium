### 3. Production Chargily Pay v2 Webhook Route (`FastAPI`)

```python
# services/api/app/routers/chargily_webhook.py
import hmac, hashlib, json, logging
from fastapi import APIRouter, Request, HTTPException, status, Header
from app.core.config import setting
from app.core.prisma import prisma

router = APIRouter(prefix="/webhooks/chargily", tags=["Billing"])
logger = logging.getLogger("billing.webhook")

@router.post("")
async def chargily_webhook(
    request: Request,
    signature: str = Header(..., alias="signature")
):
    # 1. Read Raw Body Bytes (Crucial for HMAC matching)
    raw_body = await request.body()
    secret = setting.CHARGILY_SECRET_KEY.encode("utf-8")

    # 2. Timing-safe constant-time signature verification
    computed_sig = hmac.new(secret, raw_body, hashlib.sha256).hexdigest()
    if not hmac.compare_digest(computed_sig, signature):
        logger.warning("Invalid Chargily webhook signature rejected")
        raise HTTPException(status_code=status.HTTP_403_FORBIDDEN, detail="Signature mismatch")

    payload = json.loads(raw_body.decode("utf-8"))
    event_data = payload.get("data", {})
    checkout_id = event_data.get("id")

    if payload.get("type") == "checkout.paid" and event_data.get("status") == "paid":
        user_id = event_data.get("metadata", {}).get("customer_id")
        plan_code = event_data.get("metadata", {}).get("plan_code")
        amount = event_data.get("amount")

        # 3. Atomic idempotent fulfillment transaction
        async with prisma.tx() as tx:
            existing = await tx.order.find_unique(where={"externalPaymentId": checkout_id})
            if existing and existing.status == "COMPLETED":
                return {"status": "already_processed"}

            await tx.order.upsert(
                where={"externalPaymentId": checkout_id},
                data={
                    "create": {
                        "userId": user_id,
                        "planCode": plan_code,
                        "amountDZD": amount,
                        "externalPaymentId": checkout_id,
                        "status": "COMPLETED"
                    },
                    "update": {"status": "COMPLETED"}
                }
            )
            await tx.user.update(
                where={"id": user_id},
                data={"subscriptionTier": plan_code, "aiCredits": {"increment": 5000}}
            )
        logger.info(f"Order {checkout_id} successfully fulfilled for user {user_id}")

    return {"received": True}
```

  return `https://wa.me/${cleanPhone}?text=${encodeURIComponent(message)}`;
}
